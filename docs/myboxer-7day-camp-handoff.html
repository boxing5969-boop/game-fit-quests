4.4 withInteractive 헬퍼 — 모든 체험 step 강제 모드

function withInteractive(step) {
  const remapped = SELECTOR_REMAP[step.targetSelector];
  let { selector, route } = remapped ?? { selector: step.targetSelector, route: step.route };

  if (selector !== "" && step.actionType !== "complete") {
    return {
      ...step,
      targetSelector: selector,
      route,
      actionType: "click",         // read 도 click 으로
      requireTargetClick: true,    // 직접 클릭 강제
      allowNextWithoutClick: false,
    };
  }
  return step;  // Day 완료 모달 (complete) 은 그대로
}

5. Day cooldown 정책 — 1일 1Day

completeTutorialCampDay(N<7) 호출 시:

  • currentDay = N+1, currentStep = 0 (다음 day 자리로 이동)
  • status = "paused" — 캠프 일시 정지
  • lastDayCompletedAt = ISO 시각 기록
  • day_completed 이벤트 + camp_day reflection trigger

TutorialCampProvider 마운트 시 자동 cooldown 만료 검사: 

useEffect(() => {
  if (camp.state.status !== "paused") return;
  if (!camp.state.lastDayCompletedAt) return;
  const last = new Date(camp.state.lastDayCompletedAt);
  const now = new Date();
  const sameDay =
    last.getFullYear() === now.getFullYear() &&
    last.getMonth() === now.getMonth() &&
    last.getDate() === now.getDate();
  if (!sameDay) {
    // 자정 지났음 → 자동 resume
    camp.start();
  }
});

8. 모달 자동 감지 + 자동 next 흐름

복싱 IQ / 챔피언 일기 / 응원 시트 등의 sub-기능은 회원이 모달 안에서 직접 체험. 모달 닫힘을 캠프가 자동 감지해서 다음 step 으로 자연스럽게 진입. 

// TutorialCampProvider.tsx — capture-phase listener 안
setTimeout(() => {
  if (!hasOtherDialog()) return; // 모달 안 열림 — 회원이 직접 다음 누름
  let cancelled = false;
  const id = setInterval(() => {
    if (cancelled) return;
    if (!hasOtherDialog()) {
      clearInterval(id);
      cancelled = true;
      // 모달 닫힘 → 1초 후 자동 다음
      setTimeout(() => camp.next(), 1000);
    }
  }, 800);
  setTimeout(() => { cancelled = true; clearInterval(id); }, 60_000);
}, 600);

function hasOtherDialog() {
  return Array.from(document.querySelectorAll('[role="dialog"]'))
    .some(d => !d.hasAttribute("data-tour-overlay")
            && !d.closest("[data-tour-overlay]"));
}

캠프 자체 overlay (Tooltip / Celebration) 는 data-tour-overlay="true" 마커로 식별 — 검색에서 제외해 자기 자신을 모달로 오인 방지.
